cmake_minimum_required(VERSION 3.24)
project(FunASRSubtitle VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_COMMIT 803b7fcae893e9caaee3921779628fef83ac0965)
set(LLAMA_SOURCE "${CMAKE_SOURCE_DIR}/dependencies/llama.cpp-${LLAMA_COMMIT}" CACHE PATH "Pinned llama.cpp checkout")
if(NOT EXISTS "${LLAMA_SOURCE}/ggml/CMakeLists.txt")
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 to download pinned dependencies, or set LLAMA_SOURCE")
endif()
add_subdirectory("${LLAMA_SOURCE}/ggml" ggml)
add_library(speech_engine STATIC src/sensevoice.cpp src/vad.cpp)
target_include_directories(speech_engine PUBLIC include PRIVATE vendor)
target_link_libraries(speech_engine PUBLIC ggml)
add_library(subtitle_audio STATIC src/audio.cpp)
target_include_directories(subtitle_audio PUBLIC include PRIVATE vendor)
target_link_libraries(subtitle_audio PRIVATE ole32 uuid winmm)
add_executable(FunASRSubtitle WIN32 src/app.cpp src/pipeline.cpp src/app.rc)
target_include_directories(FunASRSubtitle PRIVATE include)
target_link_libraries(FunASRSubtitle PRIVATE speech_engine subtitle_audio gdiplus shell32 comctl32)
add_executable(subtitle_bench src/bench.cpp src/pipeline.cpp)
target_include_directories(subtitle_bench PRIVATE include)
target_link_libraries(subtitle_bench PRIVATE speech_engine subtitle_audio)
add_executable(pipeline_tests tests/pipeline_tests.cpp)
add_executable(overlay_tests tests/overlay_tests.cpp)
add_dependencies(overlay_tests FunASRSubtitle)
if(MSVC)
  foreach(t FunASRSubtitle subtitle_bench)
    target_link_options(${t} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/src/app.manifest")
  endforeach()
endif()
target_include_directories(pipeline_tests PRIVATE include)
foreach(t speech_engine subtitle_audio FunASRSubtitle subtitle_bench pipeline_tests overlay_tests)
  target_compile_definitions(${t} PRIVATE NOMINMAX _USE_MATH_DEFINES UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(${t} PRIVATE /utf-8 /EHsc)
  endif()
endforeach()
enable_testing()
add_test(NAME pipeline_tests COMMAND pipeline_tests)
add_test(NAME overlay_tests COMMAND overlay_tests $<TARGET_FILE:FunASRSubtitle>)
set_tests_properties(overlay_tests PROPERTIES TIMEOUT 20)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME metrics_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_SOURCE_DIR}/tests" -p test_metrics.py -v)
endif()
